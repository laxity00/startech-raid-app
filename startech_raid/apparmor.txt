#include <tunables/global>
profile startech_raid flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  # Runtime/bootstrap access follows the official example app profile.
  file,
  signal (send,receive),
  network inet stream,
  network inet6 stream,
  capability sys_rawio,
  capability chown,
  capability dac_override,
  capability fowner,
  capability setgid,
  capability setuid,
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /tmp/** rwk,
  /data/{,**} rwk,
  /app/** r,
  /usr/local/bin/startech-raid-status rix,
  # Only the selected block device. Explicit denies override file/bootstrap rules.
  /dev/sda rk,
  deny /dev/sd* w,
  # full_access is required by Supervisor for its own system disk. Keep other
  # block devices, partition nodes and alternate SCSI/USB transports denied.
  deny /dev/sd[b-z]* rwk,
  deny /dev/sda[0-9]* rwk,
  deny /dev/sda[a-z]* rwk,
  deny /dev/{sg*,bsg/**,bus/usb/**} rwk,
  deny /dev/{mapper/**,dm-*,loop*,md*,zram*,vd*,xvd*} rwk,
  deny /dev/port rw,

  deny /dev/nvme* rw,
  deny /dev/mmcblk* rw,
  deny /dev/mem rw,
  deny /dev/kmem rw,
  deny /sys/** w,
  deny /proc/sys/** w,
}
